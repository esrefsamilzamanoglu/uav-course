cmake_minimum_required(VERSION 3.22)
project(uav_course_week2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim REQUIRED)
find_package(gz-transport REQUIRED)
find_package(gz-common REQUIRED COMPONENTS graphics)
add_library(CourseScene SHARED CourseScene.cc)
target_link_libraries(CourseScene PRIVATE gz-sim::gz-sim)
add_executable(course_camera_bridge CameraBridge.cc)
target_link_libraries(course_camera_bridge PRIVATE gz-transport::gz-transport gz-common::gz-common-graphics)
if(WIN32)
  target_compile_definitions(CourseScene PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_definitions(course_camera_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
