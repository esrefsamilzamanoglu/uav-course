cmake_minimum_required(VERSION 3.22)
project(uav_course_realistic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(gz-sim REQUIRED)
find_package(gz-transport REQUIRED)
add_library(PayloadMechanics SHARED PayloadMechanics.cc)
target_link_libraries(PayloadMechanics PRIVATE gz-sim::gz-sim gz-transport::gz-transport)
add_executable(realistic_bridge RealisticBridge.cc)
target_link_libraries(realistic_bridge PRIVATE gz-transport::gz-transport)
add_executable(follow_camera_math_test tests/follow_camera_math.cc)
target_link_libraries(follow_camera_math_test PRIVATE gz-sim::gz-sim)
enable_testing()
add_test(NAME follow_camera_geometry COMMAND follow_camera_math_test)
if(WIN32)
  target_compile_definitions(PayloadMechanics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_definitions(realistic_bridge PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(realistic_bridge PRIVATE ws2_32)
endif()
